export function escapeHTML(value) {
    const characters = {
        "&": "&amp;",
        "<": "&lt;",
        ">": "&gt;",
        '"': "&quot;",
        "'": "&#39;"
    };

    return String(value ?? "").replace(
        /[&<>"']/g,
        (character) => characters[character]
    );
}

export function renderMessage(container, message) {
    container.innerHTML = `
        <p class="empty-message">${escapeHTML(message)}</p>
    `;
}

export function formatPrice(price) {
    return `${Number(price) || 0} DH`;
}

export function formatDate(date) {
    if (!date) return "—";

    const parsedDate = new Date(`${date}T00:00:00`);

    if (Number.isNaN(parsedDate.getTime())) {
        return "—";
    }

    return new Intl.DateTimeFormat("fr-MA").format(parsedDate);
}

export function exportCSV(inscriptions) {
    const colonnes = [
        ["numeroDossier", "N° dossier"],
        ["eleve", "Élève"],
        ["niveau", "Niveau"],
        ["telephoneParent", "Téléphone parent"],
        ["cours", "Cours"],
        ["centre", "Centre"],
        ["prix", "Prix"],
        ["date", "Date"],
        ["statut", "Statut"]
    ];

    function protegerValeur(value) {
        let texte = String(value ?? "");

        if (/^[=+\-@\t\r]/.test(texte)) {
            texte = "'" + texte;
        }

        return `"${texte.replace(/"/g, '""')}"`;
    }

    const lignes = [
        colonnes.map((colonne) => protegerValeur(colonne[1])),
        ...inscriptions.map((item) =>
            colonnes.map(([cle]) => protegerValeur(item[cle]))
        )
    ];

    const contenuCSV = "\uFEFF" +
        lignes.map((ligne) => ligne.join(";")).join("\r\n");

    const fichier = new Blob(
        [contenuCSV],
        { type: "text/csv;charset=utf-8;" }
    );

    const url = URL.createObjectURL(fichier);
    const lien = document.createElement("a");

    lien.href = url;
    lien.download = "nour-academy-inscriptions.csv";
    document.body.appendChild(lien);
    lien.click();
    lien.remove();

    URL.revokeObjectURL(url);
}


========== ./js/router.js ==========
import { renderDashboard } from "./dashboard.js";
import { renderInscriptions } from "./inscriptions.js";
import { renderDetails } from "./details.js";

import {
    getState,
    getInscriptionById,
    updateInscription
} from "./state.js";

export function initRouter(container) {
    function navigate(page) {
        window.location.hash = page;
    }

    function renderPage() {
        const route = window.location.hash.slice(1) || "dashboard";

        if (route === "dashboard") {
            renderDashboard(
                container,
                getState(),
                () => navigate("inscriptions")
            );
            return;
        }

        if (route === "inscriptions") {
            renderInscriptions(
                container,
                getState(),
                (id) => navigate(`details/${id}`)
            );
            return;
        }

        if (route.startsWith("details/")) {
            const id = Number(route.split("/")[1]);
            const inscription = getInscriptionById(id);

            renderDetails(
                container,
                inscription,
                (inscriptionId, changes) => {
                    const updated = updateInscription(
                        inscriptionId,
                        changes
                    );

                    if (!updated) {
                        throw new Error("Inscription introuvable.");
                    }

                    renderPage();
                },
                () => navigate("inscriptions")
            );
            return;
        }

        navigate("dashboard");
    }

    document.querySelectorAll('nav a, .logo').forEach((link) => {
        link.addEventListener("click", (event) => {
            const href = link.getAttribute("href");

            if (href?.startsWith("#")) {
                event.preventDefault();
                navigate(href.slice(1));
            }
        });
    });

    window.addEventListener("hashchange", renderPage);

    renderPage();
}

========== ./js/validation.js ==========

export const STATUTS = [
    "En attente",
    "Validée",
    "Payée"
];

export function isValidStatus(statut) {
    return STATUTS.includes(statut);
}

export function validateInscription(inscription) {
    const erreurs = [];

    if (!inscription.eleve?.trim()) {
        erreurs.push("Le nom de l'élève est obligatoire.");
    }

    if (!inscription.niveau?.trim()) {
        erreurs.push("Le niveau est obligatoire.");
    }

    if (!isValidStatus(inscription.statut)) {
        erreurs.push("Le statut est invalide.");
    }

    return {
        valid: erreurs.length === 0,
        erreurs
    };
}


========== ./js/state.js ==========

import {
    getInscriptions,
    saveInscriptions
} from "./storage.js";

let inscriptions = getInscriptions();

export function getState() {
    return inscriptions;
}

export function setState(newInscriptions) {
    inscriptions = newInscriptions;
    saveInscriptions(inscriptions);
}

export function getInscriptionById(id) {
    return inscriptions.find((item) => item.id === Number(id));
}

export function updateInscription(id, updatedData) {
    const inscriptionExiste = inscriptions.some(
        (item) => item.id === Number(id)
    );

    if (!inscriptionExiste) {
        return false;
    }

    const nouvellesInscriptions = inscriptions.map((item) =>
        item.id === Number(id)
            ? { ...item, ...updatedData }
            : item
    );

    setState(nouvellesInscriptions);
    return true;
}

export function addInscription(inscription) {
    setState([...inscriptions, inscription]);
}

export function deleteInscription(id) {
    setState(
        inscriptions.filter((item) => item.id !== Number(id))
    );
}

import { escapeHTML, formatDate, formatPrice } from "./ui.js";
import { STATUTS } from "./validation.js";
import { exportCSV } from "./export.js";

export function renderInscriptions(container, inscriptions, onDetails) {
    container.innerHTML = `
        <section class="admin-page">
            <div class="section-heading">
                <h2>Gestion des inscriptions</h2>
                <p>Consultez et gérez les dossiers des élèves.</p>
            </div>

            <div class="admin-filters">
                <input
                    id="recherche"
                    type="search"
                    placeholder="Rechercher un élève, dossier, téléphone..."
                    aria-label="Rechercher une inscription"
                >

                <select id="filtreStatut" aria-label="Filtrer par statut">
                    <option value="">Tous les statuts</option>
                    ${STATUTS.map((statut) => `
                        <option value="${escapeHTML(statut)}">
                            ${escapeHTML(statut)}
                        </option>
                    `).join("")}
                </select>

                <button id="exporterInscriptions" type="button">
                    Exporter CSV
                </button>
            </div>

            <p id="nombreResultats" aria-live="polite"></p>
            <div id="tableInscriptions"></div>

            <button id="retourDashboard" type="button">
                Retour au Dashboard
            </button>
        </section>
    `;

    const recherche = document.getElementById("recherche");
    const filtreStatut = document.getElementById("filtreStatut");
    const tableau = document.getElementById("tableInscriptions");
    const nombreResultats = document.getElementById("nombreResultats");

    function afficherResultats() {
        const terme = recherche.value.trim().toLowerCase();
        const statutChoisi = filtreStatut.value;

        const resultats = inscriptions.filter((item) => {
            const texteRecherche = [
                item.numeroDossier,
                item.eleve,
                item.niveau,
                item.telephoneParent,
                item.cours,
                item.centre
            ].join(" ").toLowerCase();

            return (
                texteRecherche.includes(terme) &&
                (!statutChoisi || item.statut === statutChoisi)
            );
        });

        nombreResultats.textContent =
            `${resultats.length} inscription(s) trouvée(s)`;

        if (!resultats.length) {
            tableau.innerHTML = "<p>Aucune inscription trouvée.</p>";
            return;
        }

        tableau.innerHTML = `
            <div class="admin-table-wrapper">
                <table class="admin-table">
                    <thead>
                        <tr>
                            <th>N° dossier</th>
                            <th>Élève</th>
                            <th>Niveau</th>
                            <th>Téléphone parent</th>
                            <th>Centre</th>
                            <th>Cours</th>
                            <th>Prix</th>
                            <th>Date</th>
                            <th>Statut</th>
                            <th>Actions</th>
                        </tr>
                    </thead>

                    <tbody>
                        ${resultats.map((item) => `
                            <tr>
                                <td>${escapeHTML(item.numeroDossier)}</td>
                                <td>${escapeHTML(item.eleve)}</td>
                                <td>${escapeHTML(item.niveau)}</td>
                                <td>${escapeHTML(item.telephoneParent)}</td>
                                <td>${escapeHTML(item.centre)}</td>
                                <td>${escapeHTML(item.cours)}</td>
                                <td>${escapeHTML(formatPrice(item.prix))}</td>
                                <td>${escapeHTML(formatDate(item.date))}</td>
                                <td>
                                    <span class="status status-${item.statut === "Payée" ? "payee" : item.statut === "Validée" ? "validee" : "attente"}">
                                        ${escapeHTML(item.statut)}
                                    </span>
                                </td>
                                <td>
                                    <button
                                        type="button"
                                        data-details="${item.id}"
                                    >
                                        Détails
                                    </button>
                                </td>
                            </tr>
                        `).join("")}
                    </tbody>
                </table>
            </div>
        `;

        tableau.querySelectorAll("[data-details]").forEach((button) => {
            button.addEventListener("click", () => {
                onDetails(Number(button.dataset.details));
            });
        });
    }

    recherche.addEventListener("input", afficherResultats);
    filtreStatut.addEventListener("change", afficherResultats);

    document.getElementById("exporterInscriptions").addEventListener(
        "click",
        () => exportCSV(inscriptions)
    );

    document.getElementById("retourDashboard").addEventListener(
        "click",
        () => {
            window.location.hash = "dashboard";
        }
    );

    afficherResultats();
}

import { escapeHTML, formatDate, formatPrice } from "./ui.js";
import { STATUTS, isValidStatus } from "./validation.js";

export function renderDetails(
    container,
    inscription,
    onSave,
    onBack
) {
    if (!inscription) {
        container.innerHTML = `
            <section class="admin-page">
                <p>Inscription introuvable.</p>
                <button id="retourListe" type="button">
                    Retour aux inscriptions
                </button>
            </section>
        `;

        document.getElementById("retourListe").addEventListener(
            "click",
            onBack
        );
        return;
    }

    container.innerHTML = `
        <section class="admin-page">
            <div class="section-heading">
                <h2>Détails de l'inscription</h2>
                <p>Dossier ${escapeHTML(inscription.numeroDossier)}</p>
            </div>

            <div class="details-grid">
                <article class="admin-card">
                    <h3>Informations de l'élève</h3>
                    <p><strong>Nom :</strong> ${escapeHTML(inscription.eleve)}</p>
                    <p><strong>Niveau :</strong> ${escapeHTML(inscription.niveau)}</p>
                    <p><strong>Cours :</strong> ${escapeHTML(inscription.cours)}</p>
                    <p><strong>Centre :</strong> ${escapeHTML(inscription.centre)}</p>
                </article>

                <article class="admin-card">
                    <h3>Informations du dossier</h3>
                    <p><strong>Numéro :</strong> ${escapeHTML(inscription.numeroDossier)}</p>
                    <p><strong>Téléphone parent :</strong> ${escapeHTML(inscription.telephoneParent)}</p>
                    <p><strong>Date :</strong> ${escapeHTML(formatDate(inscription.date))}</p>
                    <p><strong>Prix :</strong> ${escapeHTML(formatPrice(inscription.prix))}</p>
                    <p><strong>Statut actuel :</strong> ${escapeHTML(inscription.statut)}</p>
                </article>
            </div>

            <form id="formStatut" class="details-form">
                <label for="nouveauStatut">Modifier le statut</label>

                <select id="nouveauStatut" required>
                    ${STATUTS.map((statut) => `
                        <option
                            value="${escapeHTML(statut)}"
                            ${inscription.statut === statut ? "selected" : ""}
                        >
                            ${escapeHTML(statut)}
                        </option>
                    `).join("")}
                </select>

                <button type="submit">Enregistrer le statut</button>
                <button id="retourListe" type="button">
                    Retour aux inscriptions
                </button>

                <p id="messageStatut" aria-live="polite"></p>
            </form>
        </section>
    `;

    document.getElementById("retourListe").addEventListener(
        "click",
        onBack
    );

    document.getElementById("formStatut").addEventListener(
        "submit",
        (event) => {
            event.preventDefault();

            const statut = document.getElementById("nouveauStatut").value;
            const message = document.getElementById("messageStatut");

            if (!isValidStatus(statut)) {
                message.textContent = "Veuillez choisir un statut valide.";
                return;
            }

            try {
                onSave(inscription.id, { statut });
                message.textContent = "Statut enregistré avec succès.";
            } catch (error) {
                console.error(error);
                message.textContent = "Erreur lors de l'enregistrement.";
            }
        }
    );
}

import demoData from "../data/demo-data.js";

const STORAGE_KEY = "nourAcademyInscriptions";

export function getInscriptions() {
    try {
        const savedData = localStorage.getItem(STORAGE_KEY);

        if (savedData !== null) {
            const inscriptions = JSON.parse(savedData);

            if (Array.isArray(inscriptions)) {
                return inscriptions;
            }
        }
    } catch (error) {
        console.error("Erreur de lecture des inscriptions :", error);
    }

    const initialData = demoData.map((item) => ({ ...item }));
    saveInscriptions(initialData);

    return initialData;
}

export function saveInscriptions(inscriptions) {
    try {
        localStorage.setItem(STORAGE_KEY, JSON.stringify(inscriptions));
    } catch (error) {
        console.error("Erreur de sauvegarde :", error);
        throw error;
    }
}

export function resetInscriptions() {
    const initialData = demoData.map((item) => ({ ...item }));
    saveInscriptions(initialData);

    return initialData;
}
